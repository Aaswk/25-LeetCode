## [216. 组合总和 III ](https://leetcode.cn/problems/combination-sum-iii/description/)

![image-20260316174846313](https://aaswk.oss-cn-beijing.aliyuncs.com/image-20260316174846313.png)



```c++
class Solution {
public:
    vector<int> path;
    vector<vector<int>> res;
    
    //回溯函数
    void backtracking(int sum,int n,int k,int startIndex){
        if(path.size()==k){
            if(sum==n){
                res.push_back(path);
            }//只有当满足和为n时才压入结果集，否则直接返回
            return;
        }
        
        for(int i=startIndex;i<=9;i++){
            path.push_back(i);
            sum+=i;
            backtracking(sum,n,k,i+1);//递归
            path.pop_back();//回溯
            sum-=i;//除了路径中的元素，当前的和也要回溯
            //也就是递归之前和之后是对称的
        }
    }
    
    vector<vector<int>> combinationSum3(int k, int n) {
        backtracking(0,n,k,1);
        return res;
    }
};
```

## [17. 电话号码的字母组合](https://leetcode.cn/problems/letter-combinations-of-a-phone-number/description/)

![image-20260316185400234](https://aaswk.oss-cn-beijing.aliyuncs.com/image-20260316185400234.png)

**回溯算法是用递归代替for循环**，比如说这里如果输入“23”，我就需要2个for循环，第一个循环“abc”，第二个循环“def”，对应的递归就是单层循环中先取a，然后递归取d，pop回上一层再取e

```c++
class Solution {
public:
    vector<string> res;
    string path;
    
    //建立二维数组，表示数字与字母的对应关系
    const string phoneMap[10]={"","","abc","def","ghi","jkl","mno","pqrs","tuv","wxyz"};
    
    void backtracking(string digits,int index){//index表示遍历到第几个数字
        //终止条件，数字遍历完
        if(digits.size()==index){
            res.push_back(path);
            return;
        }
        
        int digit=digits[index]-'0';//字母变数字
        string letter=phoneMap[digit];
        for(int i=0;i<letter.size();i++){
            path.push_back(letter[i]);
            backtracking(digits,index+1);
            path.pop_back();
        }
    }
    
    vector<string> letterCombinations(string digits) {
        backtracking(digits,0);
        return res;
    }
};
```
