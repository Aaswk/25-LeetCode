## [42. 接雨水 - 力扣（LeetCode）](https://leetcode.cn/problems/trapping-rain-water/description/?envType=study-plan-v2&envId=top-100-liked)

![image-20260428102600096](https://aaswk.oss-cn-beijing.aliyuncs.com/image-20260428102600096.png)



```c++
class Solution {
public:
    int trap(vector<int>& height) {
        int n=height.size();
        //pre维护前面最高的，nxt维护后面最高的
        vector<int> pre(n,0);
        vector<int> nxt(n,0);
        pre[0]=height[0];
        nxt[n-1]=height[n-1];
        for(int i=1;i<n;i++){
            pre[i]=max(height[i],pre[i-1]);
            nxt[n-i-1]=max(height[n-i-1],nxt[n-i]);
        }
        
        int res=0;
        for(int i=0;i<n;i++){
            res+=(min(pre[i],nxt[i])-height[i]);//每个桶能接多少水
        }
        return res;
    }
};
```



## [438. 找到字符串中所有字母异位词 - 力扣（LeetCode）](https://leetcode.cn/problems/find-all-anagrams-in-a-string/description/?envType=study-plan-v2&envId=top-100-liked)

![image-20260428112842893](https://aaswk.oss-cn-beijing.aliyuncs.com/image-20260428112842893.png)



```c++
class Solution {
public:
    vector<int> findAnagrams(string s, string p) {
        int k=p.size();
        int n=s.size();
        vector<int> cntp(26,0);//p字符串中每个字母出现次数
        vector<int> cnts(26,0);//当前窗口中每个字母出现次数
        for(char c:p){
            cntp[c-'a']++;
        }

        vector<int> res;
        int left=0;//定长滑窗的左边界
        for(int i=0;i<n;i++){//枚举定长滑窗右边界
            //入滑窗
            cnts[s[i]-'a']++;
            //未形成窗口
            if(i<k-1) continue;
			//出滑窗
            if(i-left+1>k){
                cnts[s[left]-'a']--;
                left++;
            }
            //加入结果集（比较是否为异位词）
            if(i-left+1==k&&cnts==cntp){
                res.push_back(left);
            }
        }
        return res;
    }
};
```
