## [40. 组合总和 II](https://leetcode.cn/problems/combination-sum-ii/description/)

![image-20260325210531177](https://aaswk.oss-cn-beijing.aliyuncs.com/image-20260325210531177.png)

1. 注意`used`数组的使用：用used数组进行树枝去重
2. 注意树枝去重和树层去重的区别

```c++
class Solution {
public:
    vector<vector<int>> res;
    vector<int> path;
    
    void backtracking(vector<int>&candidates,int target,int sum,vector<bool>&used,int startIndex){
        if(sum==target){
            res.push_back(path);
            return;
        }
        //其中的sum+candidates[i]<=target是剪枝操作
        for(int i=startIndex;i<candidates.size()&&sum+candidates[i]<=target;i++){
            //说明前一个树枝使用过candidates[i-1]
            if(i>0&&candidates[i]==candidates[i-1]&&used[i-1]==false) continue;
            //used[i-1]==true，说明同一树枝candidates[i-1]使用过
            //used[i-1]==false，说明同一树层candidates[i-1]使用过
            
            path.push_back(candidates[i]);
            sum+=candidates[i];
            used[i]=true;
            backtracking(candidates,target,sum,used,i+1);//回溯
            path.pop_back();
            sum-=candidates[i];
            used[i]=false;
        }
    }
    vector<vector<int>> combinationSum2(vector<int>& candidates, int target) {
        vector<bool> used(candidates.size(),false);
        sort(candidates.begin(),candidates.end());
        backtracking(candidates,target,0,used,0);
        return res;
    }
};
```

## [131. 分割回文串](https://leetcode.cn/problems/palindrome-partitioning/description/)

![image-20260325213551952](https://aaswk.oss-cn-beijing.aliyuncs.com/image-20260325213551952.png)

组合中[1,2,3,4]，是取1，取2，取3，取4

分割中[acb]，是在a后割，在c后割，在b后割

```c++
class Solution {
public:
    vector<vector<string>> res;
    vector<string> path;
    
    //对撞指针判断回文子串
    bool isPalindrome(string &s,int start,int end){
        while(start<=end){
            if(s[start]!=s[end]) return false;
            start++;
            end--;
        }
        return true;
    }
    
    void backtracking(string s,int startIndex){
        //切割位置在startIndex后
        if(startIndex>=s.size()){
            res.push_back(path);
            return;
        }
        
        for(int i=startIndex;i<s.size();i++){
            if(isPalindrome(s,startIndex,i)){
                //substr用法：起始索引和截取长度
                string str=s.substr(startIndex,i-startIndex+1);
                
                path.push_back(str);
                backtracking(s,i+1);
                path.pop_back();
            }
        }
    }
    vector<vector<string>> partition(string s) {
        backtracking(s,0);
        return res;
    }
};
```
