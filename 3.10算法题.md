## [15. 三数之和](https://leetcode.cn/problems/3sum/description/)

![image-20260310213546465](https://aaswk.oss-cn-beijing.aliyuncs.com/image-20260310213546465.png)

排序+双指针

```c++
class Solution {
public:
    vector<vector<int>> threeSum(vector<int>& nums) {
        vector<vector<int>> res;
        int n=nums.size();
        //1.排序：方便去重（二分查找必须有序）
        sort(nums.begin(),nums.end());
        //2.固定一个数k，双指针寻找nums[left]+nums[right]=-nums[k]
        for(int k=0;k<n-2;k++){
            if(k>0&&nums[k]==nums[k-1]) continue;//去重
            
            //二分查找
            int left=k+1;
            int right=n-1;
            while(left<right){
                int sum=nums[left]+nums[right]+nums[k];
                if(sum>0) right--;
                else if(sum<0) left++;
                else {
                    res.push_back({nums[k],nums[left],nums[right]});//找到满足题意的三元组
                    left++;
                    right--;
                    while(nums[left]==nums[left-1]&&left<right) left++;//去重
                    while(nums[right]==nums[right+1]&&left<right) right--;//去重
                }
            }
        }
        return res;
    }
};

```

## [18. 四数之和](https://leetcode.cn/problems/4sum/description/)

![image-20260310215638751](https://aaswk.oss-cn-beijing.aliyuncs.com/image-20260310215638751.png)

排序+双指针+双层for循环

```c++
class Solution {
public:
    vector<vector<int>> fourSum(vector<int>& nums, int target) {
        vector<vector<int>> res;
        int n=nums.size();
        //1.排序
        sort(nums.begin(),nums.end());
        
        //2.通过双层for循环实现每次查找时固定了2个数
        for(int i=0;i<n-2;i++){
            if(i>0&&nums[i]==nums[i-1]) continue;
            for(int j=i+1;j<n-1;j++){
            if(j>i+1&&nums[j]==nums[j-1]) continue;
                
                //二分查找
                int left=j+1;
                int right=n-1;
                while(left<right){
                    //注意数的大小
                    long long sum=(long long)nums[i]+nums[j]+nums[left]+nums[right];
                    if(sum<target) left++;
                    else if(sum>target) right--;
                    else{
                        res.push_back({nums[i],nums[j],nums[left],nums[right]});
                        left++;
                        right--;
                        while(left<right&&nums[left]==nums[left-1]) left++;
                        while(left<right&&nums[right]==nums[right+1]) right--;
                    }
                }
            }
        }
        return res;
    }
};
```
