## [862. 和至少为 K 的最短子数组 - 力扣（LeetCode）](https://leetcode.cn/problems/shortest-subarray-with-sum-at-least-k/description/?envType=problem-list-v2&envId=monotonic-queue)

![image-20260414153026197](https://aaswk.oss-cn-beijing.aliyuncs.com/image-20260414153026197.png)



```c++
class Solution {
public:
    int shortestSubarray(vector<int>& nums, int k) {
        int ans=nums.size()+1;
        //构造前缀和
        long long prefix[nums.size()+1];
        prefix[0]=0;
        for(int i=1;i<=nums.size();i++){
            prefix[i]=prefix[i-1]+nums[i-1];
        }
        //单调队列
        deque<long long> dq;
        for(int i=0;i<=nums.size();i++){
	        //队尾添加元素（维持单调性）
            while(!dq.empty()&&prefix[i]<=prefix[dq.back()]){
                dq.pop_back();
            }
            dq.push_back(i); 
             
			//根据队头元素计算结果
            while(!dq.empty()&&prefix[i]-prefix[dq.front()]>=k){
                int left=dq.front();
                dq.pop_front();
                ans=min(ans,i-left);
            }
        }
        
        //出答案了
        if(ans>nums.size()){
            return -1;
        }
        return ans;
    }
};
```

## [918. 环形子数组的最大和 - 力扣（LeetCode）](https://leetcode.cn/problems/maximum-sum-circular-subarray/description/?envType=problem-list-v2&envId=monotonic-queue)

![image-20260414213745111](https://aaswk.oss-cn-beijing.aliyuncs.com/image-20260414213745111.png)



```c++
class Solution {
public:
    int maxSubarraySumCircular(vector<int>& nums) {
        int n = nums.size();
        // 将数组复制一倍
        vector<int> arr = nums;
        arr.insert(arr.end(), nums.begin(), nums.end());
        // 前缀和数组，长度为 2n+1
        vector<int> pref(2 * n + 1, 0);
        for (int i = 1; i <= 2 * n; ++i) {
            pref[i] = pref[i - 1] + arr[i - 1];
        }
        deque<int> dq;  // 存储前缀和的索引，保持单调递增（值递增）
        int ans = INT_MIN;
        for (int i = 0; i <= 2 * n; ++i) {
            // 移除超出窗口范围（窗口大小为 n）的索引
            while (!dq.empty() && dq.front() < i - n) {
                dq.pop_front();
            }
            // 当 i > 0 时，计算以 i-1 结尾的子数组和
            if (i > 0 && !dq.empty()) {
                int cur = pref[i] - pref[dq.front()];
                if (cur > ans) ans = cur;
            }
            // 维护单调递增队列：队尾前缀和 >= 当前前缀和则弹出
            while (!dq.empty() && pref[dq.back()] >= pref[i]) {
                dq.pop_back();
            }
            dq.push_back(i);
        }
        return ans;
    }
};
```
