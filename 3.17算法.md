## [404. 左叶子之和](https://leetcode.cn/problems/sum-of-left-leaves/description/)

![image-20260317190633657](https://aaswk.oss-cn-beijing.aliyuncs.com/image-20260317190633657.png)

即可以将sum作为参数（要传引用），也可以作为返回值。

如果作为返回值，就需要sum+=递归函数的形式来接。

第一种写法：

```c++
class Solution {
public:
    void back(TreeNode*cur,int& sum){
        if(cur==nullptr) return;
        //因为要保证左叶子为空，所以通过判断上一节点来确保是“左”
        if(cur->left!=nullptr&&cur->left->left==nullptr&&cur->left->right==nullptr){
            sum+=cur->left->val;
        }
        back(cur->left,sum);
        back(cur->right,sum);
    }
    
    int sumOfLeftLeaves(TreeNode* root) {
        int sum=0;
        back(root,sum);
        return sum;
    }
};
```

第二种写法：

```c++
class Solution {
public:
    int sumOfLeftLeaves(TreeNode* root) {
        int sum=0;
        if(root==nullptr) return sum;
        if(root->left!=nullptr&&root->left->left==nullptr&&root->left->right==nullptr){
            sum+=root->left->val;
        }
        sum+=sumOfLeftLeaves(root->left);
        sum+=sumOfLeftLeaves(root->right);
        return sum;
    }
};
```

## [39. 组合总和](https://leetcode.cn/problems/combination-sum/description/)

![image-20260317192805088](https://aaswk.oss-cn-beijing.aliyuncs.com/image-20260317192805088.png)

```c++
class Solution {
public:
    vector<vector<int>> res;
    vector<int> path;
    
    void backtracking(vector<int>&candidates,int target,int startIndex,int sum){
        if(sum>target){
            return;
        }
        if(sum==target){
            res.push_back(path);
        }
        //在同一集合中选取元素用startIndex表示
        for(int i=startIndex;i<candidates.size();i++){
            sum+=candidates[i];
            path.push_back(candidates[i]);
            backtracking(candidates,target,i,sum);//使用i，可以重复读取当前的数
            sum-=candidates[i];
            path.pop_back();
        }
        //调用回溯函数前后是对称的，添加了什么操作就要撤回什么操作
    }
    vector<vector<int>> combinationSum(vector<int>& candidates, int target) {
        backtracking(candidates,target,0,0);
        return res;
    }
};
```
